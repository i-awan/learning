from flask import Flask, request, send_from_directory
import subprocess

app = Flask(__name__)

AUTH_PASSWORD = "supersecret"  # hardcoded for simplicity

@app.route('/')
def form():
    return send_from_directory('.', 'index.html')

@app.route('/trigger', methods=['POST'])
def trigger():
    username = request.form.get('username')
    password = request.form.get('password')

    if password != AUTH_PASSWORD:
        return "Unauthorized", 403

    curl_command = [
        'curl', '-X', 'POST', 'https://external-api.com/endpoint',
        '-H', 'Content-Type: application/json',
        '--data-raw', f'''{{
            "input_token_state": {{
                "token_type": "CREDENTIAL",
                "username": "{username}",
                "password": "{password}"
            }}
        }}'''
    ]

    try:
        result = subprocess.check_output(curl_command, stderr=subprocess.STDOUT)
        return f"<pre>{result.decode()}</pre>"
    except subprocess.CalledProcessError as e:
        return f"<pre>Failed:\n{e.output.decode()}</pre>", 500

from flask import Flask, request, send_from_directory, render_template_string

app = Flask(__name__)

AUTH_PASSWORD = "supersecret"  # Hardcoded password

MATCH_TEMPLATE = """
<!DOCTYPE html>
<html>
<head><title>Match</title></head>
<body>
  <h2>✅ Password is a match</h2>
  <a href="/">Go back</a>
</body>
</html>
"""

NO_MATCH_TEMPLATE = """
<!DOCTYPE html>
<html>
<head><title>No Match</title></head>
<body>
  <h2>❌ Password is incorrect</h2>
  <a href="/">Try again</a>
</body>
</html>
"""

@app.route('/')
def form():
    return send_from_directory('.', 'index.html')

@app.route('/trigger', methods=['POST'])
def trigger():
    password = request.form.get('password')

    if password == AUTH_PASSWORD:
        return render_template_string(MATCH_TEMPLATE)
    else:
        return render_template_string(NO_MATCH_TEMPLATE), 403



- name: Upload file to S3 using a pre-signed URL
  ansible.builtin.uri:
    url: "https://your-s3-endpoint.com/my-bucket/myfile.txt?AWSAccessKeyId=...&Signature=..."
    method: PUT
    src: "/path/to/myfile.txt"
    headers:
      Content-Type: "application/octet-stream"


import boto3

s3 = boto3.client(
    's3',
    endpoint_url='https://your-s3-endpoint.com',
    aws_access_key_id='YOUR_ACCESS_KEY',
    aws_secret_access_key='YOUR_SECRET_KEY',
)

url = s3.generate_presigned_url(
    ClientMethod='put_object',
    Params={
        'Bucket': 'my-bucket',
        'Key': 'myfile.txt',
    },
    ExpiresIn=3600  # URL valid for 1 hour
)

print(url)
