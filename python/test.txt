import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin

session = requests.Session()

login_page_url = "https://example.com/login"   # the page you visit in the browser
protected_url   = "https://example.com/account"

# 1) GET login page
r = session.get(login_page_url)
r.raise_for_status()

soup = BeautifulSoup(r.text, "html.parser")

# 2) Find the <form> and its action/method
form = soup.find("form")
action = form.get("action")
method = (form.get("method") or "get").lower()

login_post_url = urljoin(login_page_url, action)  # handle relative URLs like "/auth/login"

# 3) Extract CSRF token (adjust name as needed)
csrf_input = form.find("input", {"name": "csrf_token"})
csrf_token = csrf_input["value"]

# 4) Build payload
payload = {
    "username": "your_username",
    "password": "your_password",
    "csrf_token": csrf_token,
}

headers = {
    "User-Agent": "Mozilla/5.0",
    "Referer": login_page_url,
}

# 5) Send request using the form's method
if method == "post":
    login_resp = session.post(login_post_url, data=payload, headers=headers)
else:
    login_resp = session.get(login_post_url, params=payload, headers=headers)

login_resp.raise_for_status()
print("Logged in? Final URL:", login_resp.url)

# 6) Get protected page
protected_resp = session.get(protected_url, headers=headers)
protected_resp.raise_for_status()
print(protected_resp.text)
